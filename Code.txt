/**
 * Kalinga sa Gala – Backend (Google Apps Script)
 * Nag-iimbak ng pet listings sa Google Sheets at ng mga litrato sa Google Drive.
 *
 * API (ginagamit ng index.html):
 *   GET  ?action=list&callback=fn[&token=...]   -> JSONP { success, pets, history }
 *   POST { action:'login', password }           -> { success, token }
 *   POST { action:'logout', token }             -> { success }
 *   POST { action:'addPet', token, name, type, otherType, gender, description, photoBase64 }
 *   POST { action:'adoptPet', token, id }
 *   POST { action:'deletePet', token, id }   -> binubura ang row at ang litrato
 *
 * SETUP: tingnan ang README.md
 */

var BACKEND_VERSION = '3';   // lumalabas sa ?action=list para makumpirma kung bago ang deployed code
var SHEET_NAME   = 'Pets';
var HEADERS      = ['id','name','type','otherType','gender','description','photoFileId','status','createdAt','adoptedAt'];
var TOKEN_TTL    = 21600;   // 6 oras (max ng CacheService)
var MAX_FAILS    = 5;       // maling login bago i-lock
var LOCK_SECONDS = 600;     // 10 minutong lock
var MAX_PHOTO_BYTES = 5 * 1024 * 1024;
var PHOTO_TYPES  = ['image/jpeg','image/png','image/webp','image/gif'];

/* ---------- ONE-TIME SETUP (i-run mo ito nang manu-mano) ---------- */
function setup() {
  var props = PropertiesService.getScriptProperties();
  if (!props.getProperty('ADMIN_PASSWORD')) {
    throw new Error('Magdagdag muna ng Script Property na ADMIN_PASSWORD (Project Settings > Script Properties), tapos i-run ulit ang setup().');
  }

  var ss = null;
  var ssId = props.getProperty('SHEET_ID');
  if (ssId) { try { ss = SpreadsheetApp.openById(ssId); } catch (e) { ss = null; } }
  if (!ss) {
    ss = SpreadsheetApp.create('Kalinga sa Gala – Pets');
    props.setProperty('SHEET_ID', ss.getId());
  }
  var sh = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
  // Plain text para hindi ma-interpret na formula ang input ng user
  sh.getRange(1, 1, sh.getMaxRows(), HEADERS.length).setNumberFormat('@');
  sh.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]).setFontWeight('bold');
  sh.setFrozenRows(1);
  var def = ss.getSheetByName('Sheet1');
  if (def && ss.getSheets().length > 1) ss.deleteSheet(def);

  var folder = null;
  var fId = props.getProperty('FOLDER_ID');
  if (fId) { try { folder = DriveApp.getFolderById(fId); } catch (e) { folder = null; } }
  if (!folder) {
    folder = DriveApp.createFolder('KalingaSaGala_Photos');
    props.setProperty('FOLDER_ID', folder.getId());
  }

  Logger.log('OK! Sheet: ' + ss.getUrl());
  Logger.log('Photos folder: ' + folder.getUrl());
}

/* ---------- ENTRY POINTS ---------- */
function doGet(e) {
  var p = (e && e.parameter) || {};
  var out;
  try {
    if (!p.action || p.action === 'list') out = listPets_(p.token);
    else out = fail_('Hindi kilalang action.');
  } catch (err) { out = handleError_(err); }
  return respond_(out, p.callback);
}

function doPost(e) {
  var out;
  try {
    var raw = (e && e.parameter && e.parameter.payload) || (e && e.postData && e.postData.contents) || '';
    var data;
    try { data = JSON.parse(raw); } catch (x) { throw new UserError('Sira ang request na natanggap.'); }
    switch (data.action) {
      case 'login':    out = login_(data); break;
      case 'logout':   out = logout_(data); break;
      case 'addPet':   out = withLock_(function () { return addPet_(data); }); break;
      case 'adoptPet': out = withLock_(function () { return adoptPet_(data); }); break;
      case 'deletePet': out = withLock_(function () { return deletePet_(data); }); break;
      default:         out = fail_('Hindi kilalang action.');
    }
  } catch (err) { out = handleError_(err); }
  return respond_(out, null);
}

/* ---------- ACTIONS ---------- */
function listPets_(token) {
  var isAdmin = !!token && isValidToken_(token);
  var rows = readRows_();
  var pets = [], history = [];
  rows.forEach(function (r) {
    if (r.status === 'adopted') history.push(toPublic_(r));
    else pets.push(toPublic_(r));
  });
  return { success: true, version: BACKEND_VERSION, pets: pets, history: isAdmin ? history : [] };
}

function login_(d) {
  var cache = CacheService.getScriptCache();
  if (Number(cache.get('login_fails') || 0) >= MAX_FAILS) {
    return fail_('Masyadong maraming maling subok. Subukan ulit pagkalipas ng 10 minuto.');
  }
  var expected = PropertiesService.getScriptProperties().getProperty('ADMIN_PASSWORD');
  if (!expected) throw new UserError('Wala pang ADMIN_PASSWORD sa backend. Tingnan ang setup.');
  if (!safeEqual_(String(d.password || ''), expected)) {
    var n = Number(cache.get('login_fails') || 0) + 1;
    cache.put('login_fails', String(n), LOCK_SECONDS);
    return fail_('Maling password.');
  }
  cache.remove('login_fails');
  var token = Utilities.getUuid() + Utilities.getUuid();
  cache.put('tok_' + token, '1', TOKEN_TTL);
  return { success: true, token: token };
}

function logout_(d) {
  if (d.token) CacheService.getScriptCache().remove('tok_' + d.token);
  return { success: true };
}

function addPet_(d) {
  requireAdmin_(d.token);

  var name = clean_(d.name, 40);
  var type = String(d.type || '');
  var other = clean_(d.otherType, 40);
  var gender = String(d.gender || '');
  var desc = clean_(d.description, 500);

  if (!name || !desc) return fail_('Kulang ang pangalan o description.');
  if (['dog', 'cat', 'other'].indexOf(type) < 0) return fail_('Hindi valid ang type ng hayop.');
  if (['male', 'female', 'unknown'].indexOf(gender) < 0) return fail_('Hindi valid ang gender.');
  if (type === 'other' && !other) return fail_('Ilagay kung anong hayop (Other).');
  if (type !== 'other') other = '';

  var id = Utilities.getUuid();
  var fileId = '';
  if (d.photoBase64) fileId = savePhoto_(d.photoBase64, name, id);

  var row = [id, name, type, other, gender, desc, fileId, 'available', new Date().toISOString(), ''];
  var sh = getSheet_();
  var target = sh.getLastRow() + 1;
  sh.getRange(target, 1, 1, HEADERS.length).setNumberFormat('@').setValues([row]);
  return { success: true, id: id };
}

function adoptPet_(d) {
  requireAdmin_(d.token);
  var sh = getSheet_();
  var last = sh.getLastRow();
  if (last < 2) return fail_('Walang nakitang hayop.');
  var ids = sh.getRange(2, 1, last - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) {
    if (String(ids[i][0]) === String(d.id)) {
      var rowNum = i + 2;
      var status = sh.getRange(rowNum, 8).getValue();
      if (status === 'adopted') return fail_('Naka-adopt na ang hayop na ito.');
      sh.getRange(rowNum, 8, 1, 3).setValues([['adopted', sh.getRange(rowNum, 9).getValue(), new Date().toISOString()]]);
      return { success: true };
    }
  }
  return fail_('Hindi nahanap ang hayop.');
}

function deletePet_(d) {
  requireAdmin_(d.token);
  var sh = getSheet_();
  var last = sh.getLastRow();
  if (last < 2) return fail_('Walang nakitang hayop.');
  var ids = sh.getRange(2, 1, last - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) {
    if (String(ids[i][0]) === String(d.id)) {
      var rowNum = i + 2;
      var fileId = String(sh.getRange(rowNum, 7).getValue() || '');
      sh.deleteRow(rowNum);
      if (fileId) { try { DriveApp.getFileById(fileId).setTrashed(true); } catch (e) { console.warn('photo trash failed: ' + e); } }
      return { success: true };
    }
  }
  return fail_('Hindi nahanap ang hayop.');
}

/* ---------- HELPERS ---------- */
function savePhoto_(input, name, id) {
  var s = String(input);
  var mime = 'image/jpeg';
  var b64 = s;
  var m = s.match(/^data:([\w\/+.-]+);base64,/);   // tinatanggap ang buong data URL galing sa frontend
  if (m) { mime = m[1].toLowerCase(); b64 = s.slice(m[0].length); }
  if (PHOTO_TYPES.indexOf(mime) < 0) throw new UserError('JPG, PNG, WEBP, o GIF lang ang pwedeng litrato.');
  var bytes;
  try { bytes = Utilities.base64Decode(b64); } catch (e) { throw new UserError('Sira ang litrato. Subukan ang ibang file.'); }
  if (bytes.length > MAX_PHOTO_BYTES) throw new UserError('Masyadong malaki ang litrato (max 5 MB).');

  var ext = mime.split('/')[1].replace('jpeg', 'jpg');
  var safeName = name.replace(/[^a-z0-9_-]/gi, '_').slice(0, 30) || 'pet';
  var blob = Utilities.newBlob(bytes, mime, safeName + '_' + id.slice(0, 8) + '.' + ext);
  var file = getFolder_().createFile(blob);
  try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) { console.warn('setSharing failed: ' + e); }
  return file.getId();
}

function readRows_() {
  var sh = getSheet_();
  var last = sh.getLastRow();
  if (last < 2) return [];
  return sh.getRange(2, 1, last - 1, HEADERS.length).getValues().map(function (v) {
    var o = {};
    HEADERS.forEach(function (h, i) { o[h] = v[i] === null || v[i] === undefined ? '' : String(v[i]); });
    return o;
  }).filter(function (o) { return o.id; });
}

function toPublic_(r) {
  return {
    id: r.id,
    name: r.name,
    type: r.type,
    otherType: r.otherType,
    gender: r.gender,
    description: r.description,
    photoUrl: r.photoFileId ? 'https://drive.google.com/thumbnail?id=' + encodeURIComponent(r.photoFileId) + '&sz=w800' : '',
    adoptedAt: r.adoptedAt || ''
  };
}

function getSheet_() {
  var id = PropertiesService.getScriptProperties().getProperty('SHEET_ID');
  if (!id) throw new UserError('Hindi pa na-run ang setup() sa Apps Script.');
  var sh = SpreadsheetApp.openById(id).getSheetByName(SHEET_NAME);
  if (!sh) throw new UserError('Walang "' + SHEET_NAME + '" sheet. I-run ulit ang setup().');
  return sh;
}

function getFolder_() {
  var id = PropertiesService.getScriptProperties().getProperty('FOLDER_ID');
  if (!id) throw new UserError('Hindi pa na-run ang setup() sa Apps Script.');
  return DriveApp.getFolderById(id);
}

function isValidToken_(t) { return !!t && CacheService.getScriptCache().get('tok_' + t) === '1'; }

function requireAdmin_(t) {
  if (!isValidToken_(t)) {
    var err = new UserError('Nag-expire ang admin session. Mag-login ulit.');
    err.code = 'AUTH';
    throw err;
  }
}

function withLock_(fn) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try { return fn(); } finally { lock.releaseLock(); }
}

function clean_(v, max) {
  return String(v == null ? '' : v).replace(/[\u0000-\u001F\u007F]/g, ' ').trim().slice(0, max);
}

function safeEqual_(a, b) {
  var ha = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, a);
  var hb = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, b);
  var diff = 0;
  for (var i = 0; i < ha.length; i++) diff |= (ha[i] ^ hb[i]);
  return diff === 0;
}

function UserError(message) { this.message = message; this.userMessage = true; }
UserError.prototype = Object.create(Error.prototype);

function fail_(message) { return { success: false, message: message }; }

function handleError_(err) {
  if (err && err.userMessage) return { success: false, message: err.message, code: err.code || '' };
  console.error(err && err.stack ? err.stack : err);
  return fail_('May problema sa server. Subukan ulit mamaya.');
}

function respond_(obj, callback) {
  var json = JSON.stringify(obj);
  if (callback && /^[A-Za-z_$][\w$.]{0,80}$/.test(callback)) {
    return ContentService.createTextOutput(callback + '(' + json + ');').setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}
